cmake_minimum_required(VERSION 3.20)
project(staggerlab VERSION 0.1.0 LANGUAGES CXX)
option(STAGGERLAB_PYTHON "Build Python bindings" ON)
include(CTest)
add_library(operators INTERFACE)
target_include_directories(operators INTERFACE cpp)
target_compile_features(operators INTERFACE cxx_std_17)
if(MSVC)
  target_compile_options(operators INTERFACE /W4 /WX)
else()
  target_compile_options(operators INTERFACE -Wall -Wextra -Wpedantic -Werror)
endif()
if(STAGGERLAB_PYTHON)
  find_package(Python COMPONENTS Interpreter Development.Module REQUIRED)
  find_package(pybind11 CONFIG REQUIRED)
  pybind11_add_module(_core cpp/bindings.cpp)
  target_link_libraries(_core PRIVATE operators)
  install(TARGETS _core LIBRARY DESTINATION staggerlab RUNTIME DESTINATION staggerlab)
endif()
if(BUILD_TESTING)
  add_executable(test_operators cpp/test_operators.cpp)
  target_link_libraries(test_operators PRIVATE operators)
  add_test(NAME compatible_operators COMMAND test_operators)
endif()
